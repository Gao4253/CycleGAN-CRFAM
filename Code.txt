import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F
import torchvision.models as models

# ==========================================
# 1. Network Components
# ==========================================

class SelfAttention(nn.Module):
    """
    Self-Attention Module (SAM) for global style consistency.
    Equation (1) and (2) from the paper.
    """
    def __init__(self, in_dim):
        super(SelfAttention, self).__init__()
        self.query_conv = nn.Conv2d(in_dim, in_dim // 8, 1)
        self.key_conv = nn.Conv2d(in_dim, in_dim // 8, 1)
        self.value_conv = nn.Conv2d(in_dim, in_dim, 1)
        # Learnable parameter initialized to 0
        self.gamma = nn.Parameter(torch.zeros(1))
        self.softmax = nn.Softmax(dim=-1)

    def forward(self, x):
        m_batchsize, C, width, height = x.size()
        
        # Q, K, V projections
        proj_query = self.query_conv(x).view(m_batchsize, -1, width * height).permute(0, 2, 1)
        proj_key = self.key_conv(x).view(m_batchsize, -1, width * height)
        
        # Attention map calculation
        energy = torch.bmm(proj_query, proj_key)
        attention = self.softmax(energy)
        
        proj_value = self.value_conv(x).view(m_batchsize, -1, width * height)
        out = torch.bmm(proj_value, attention.permute(0, 2, 1))
        out = out.view(m_batchsize, C, width, height)
        
        # Weighted output
        out = self.gamma * out + x
        return out

class CRFAM(nn.Module):
    """
    Cultural-Relic Feature-Attention Module.
    Generates a spatial attention map to protect key structural regions.
    """
    def __init__(self, in_channels):
        super(CRFAM, self).__init__()
        # Micro-CNN: Conv1x1 -> Depthwise Conv3x3 -> Conv1x1 -> Sigmoid
        self.conv1 = nn.Conv2d(in_channels, in_channels // 8, 1)
        self.dw_conv = nn.Conv2d(in_channels // 8, in_channels // 8, 3, padding=1, groups=in_channels // 8)
        self.conv2 = nn.Conv2d(in_channels // 8, 1, 1)
        self.sigmoid = nn.Sigmoid()

    def forward(self, x):
        x_compress = torch.relu(self.conv1(x))
        x_spatial = torch.relu(self.dw_conv(x_compress))
        attn_map = self.sigmoid(self.conv2(x_spatial))
        
        # Equation (3): Element-wise multiplication
        weighted_features = x * attn_map
        return attn_map, weighted_features

class ChannelMLP(nn.Module):
    """
    Structurally Guided Channel MLP.
    Filters and enhances channel features based on spatial attention.
    """
    def __init__(self, in_channels, reduction=8):
        super(ChannelMLP, self).__init__()
        self.mlp = nn.Sequential(
            nn.Linear(in_channels, in_channels // reduction),
            nn.ReLU(inplace=True),
            nn.Linear(in_channels // reduction, in_channels),
            nn.Sigmoid()
        )

    def forward(self, x, attn_map):
        B, C, H, W = x.size()
        
        # Equation (4): Weighted pooling using attention map
        attn_sum = torch.sum(attn_map.view(B, 1, -1), dim=2, keepdim=True) + 1e-8
        A_tilde = attn_map.view(B, 1, -1) / attn_sum 
        
        F_prime = x.view(B, C, -1) 
        desc = torch.sum(F_prime * A_tilde, dim=2) # (B, C)
        
        # Channel weights via MLP
        weight = self.mlp(desc).view(B, C, 1, 1)
        
        # Channel-wise multiplication
        return x * weight

class ResnetBlock(nn.Module):
    """Residual block with embedded Self-Attention Module."""
    def __init__(self, dim):
        super(ResnetBlock, self).__init__()
        self.conv_block = nn.Sequential(
            nn.ReflectionPad2d(1),
            nn.Conv2d(dim, dim, 3),
            nn.InstanceNorm2d(dim),
            nn.ReLU(True),
            nn.ReflectionPad2d(1),
            nn.Conv2d(dim, dim, 3),
            nn.InstanceNorm2d(dim)
        )
        self.sam = SelfAttention(dim)

    def forward(self, x):
        out = self.conv_block(x)
        out = self.sam(out)
        return x + out

class Generator(nn.Module):
    """CycleGAN Generator with CRFAM, Channel MLP, and SAM."""
    def __init__(self, input_nc=3, output_nc=3, ngf=64, n_blocks=6):
        super(Generator, self).__init__()
        
        # Encoder (Downsampling)
        model = [
            nn.ReflectionPad2d(3),
            nn.Conv2d(input_nc, ngf, 7),
            nn.InstanceNorm2d(ngf),
            nn.ReLU(True)
        ]
        curr_dim = ngf
        for i in range(2):
            model += [
                nn.Conv2d(curr_dim, curr_dim * 2, 3, stride=2, padding=1),
                nn.InstanceNorm2d(curr_dim * 2),
                nn.ReLU(True)
            ]
            curr_dim *= 2
        self.encoder = nn.Sequential(*model)

        # CRFAM and Channel MLP
        self.crfam = CRFAM(curr_dim)
        self.channel_mlp = ChannelMLP(curr_dim)

        # Bottleneck with SAM
        res_blocks = [ResnetBlock(curr_dim) for _ in range(n_blocks)]
        self.bottleneck = nn.Sequential(*res_blocks)

        # Decoder (Upsampling)
        model = []
        for i in range(2):
            model += [
                nn.Upsample(scale_factor=2, mode='nearest'),
                nn.Conv2d(curr_dim, curr_dim // 2, 3, stride=1, padding=1),
                nn.InstanceNorm2d(curr_dim // 2),
                nn.ReLU(True)
            ]
            curr_dim //= 2
        model += [
            nn.ReflectionPad2d(3),
            nn.Conv2d(curr_dim, output_nc, 7),
            nn.Tanh()
        ]
        self.decoder = nn.Sequential(*model)

    def forward(self, x):
        # Paper sequence: Encoder -> CRFAM -> Channel MLP -> Bottleneck (SAM) -> Decoder
        x_enc = self.encoder(x)
        attn_map, x_weighted = self.crfam(x_enc)
        x_refined = self.channel_mlp(x_weighted, attn_map)
        x_bot = self.bottleneck(x_refined)
        out = self.decoder(x_bot)
        return out, attn_map

class Discriminator(nn.Module):
    """Standard PatchGAN Discriminator."""
    def __init__(self, input_nc=3, ndf=64):
        super(Discriminator, self).__init__()
        model = [
            nn.Conv2d(input_nc, ndf, 4, stride=2, padding=1),
            nn.LeakyReLU(0.2, True)
        ]
        model += [
            nn.Conv2d(ndf, ndf * 2, 4, stride=2, padding=1),
            nn.InstanceNorm2d(ndf * 2),
            nn.LeakyReLU(0.2, True)
        ]
        model += [
            nn.Conv2d(ndf * 2, ndf * 4, 4, stride=2, padding=1),
            nn.InstanceNorm2d(ndf * 4),
            nn.LeakyReLU(0.2, True)
        ]
        model += [
            nn.Conv2d(ndf * 4, ndf * 8, 4, stride=1, padding=1),
            nn.InstanceNorm2d(ndf * 8),
            nn.LeakyReLU(0.2, True)
        ]
        model += [nn.Conv2d(ndf * 8, 1, 4, stride=1, padding=1)]
        self.model = nn.Sequential(*model)

    def forward(self, x):
        return self.model(x)

class VGG16FeatureExtractor(nn.Module):
    """Pretrained VGG16 for Attention-Guided Loss."""
    def __init__(self):
        super(VGG16FeatureExtractor, self).__init__()
        vgg = models.vgg16(weights=models.VGG16_Weights.IMAGENET1K_V1).features
        # Extract features up to relu3_3 (index 15) for mid-level structural features
        self.features = nn.Sequential(*list(vgg.children())[:16])
        for param in self.features.parameters():
            param.requires_grad = False

    def forward(self, x):
        return self.features(x)

# 2. Main Framework
# ==========================================

class AttentionCycleGAN(nn.Module):
    def __init__(self, lambda_cyc=10.0, lambda_attn=5.0):
        super(AttentionCycleGAN, self).__init__()
        # Generators
        self.G_A2B = Generator()
        self.G_B2A = Generator()
        # Discriminators
        self.D_A = Discriminator()
        self.D_B = Discriminator()
        # Feature extractor for attention loss
        self.vgg = VGG16FeatureExtractor()

        # Loss functions
        self.criterion_GAN = nn.MSELoss() # LSGAN
        self.criterion_cyc = nn.L1Loss()
        self.criterion_attn = nn.MSELoss()

        # Hyperparameters
        self.lambda_cyc = lambda_cyc
        self.lambda_attn = lambda_attn

    def train_step(self, real_A, real_B, optimizer_G, optimizer_D):
        
        # ------------------
        # Generators Forward
        # ------------------
        fake_B, attn_A = self.G_A2B(real_A)
        rec_A, _ = self.G_B2A(fake_B)
        
        fake_A, attn_B = self.G_B2A(real_B)
        rec_B, _ = self.G_A2B(fake_A)
        
        # ------------------
        # Generator Step
        # ------------------
        optimizer_G.zero_grad()
        
        # Adversarial Loss
        pred_fake_B = self.D_B(fake_B)
        loss_GAN_A2B = self.criterion_GAN(pred_fake_B, torch.ones_like(pred_fake_B))
        
        pred_fake_A = self.D_A(fake_A)
        loss_GAN_B2A = self.criterion_GAN(pred_fake_A, torch.ones_like(pred_fake_A))
        loss_GAN = (loss_GAN_A2B + loss_GAN_B2A) * 0.5
        
        # Cycle Consistency Loss
        loss_cyc_A = self.criterion_cyc(rec_A, real_A)
        loss_cyc_B = self.criterion_cyc(rec_B, real_B)
        loss_cyc = (loss_cyc_A + loss_cyc_B) * 0.5
        
        # Attention-Guided Loss
        # Upsample attention map to VGG feature resolution
        feat_real_A = self.vgg(real_A)
        feat_fake_B = self.vgg(fake_B)
        attn_A_up = F.interpolate(attn_A, size=feat_real_A.shape[2:], mode='bilinear', align_corners=False)
        loss_attn_A = self.criterion_attn(attn_A_up * feat_real_A, attn_A_up * feat_fake_B)
        
        feat_real_B = self.vgg(real_B)
        feat_fake_A = self.vgg(fake_A)
        attn_B_up = F.interpolate(attn_B, size=feat_real_B.shape[2:], mode='bilinear', align_corners=False)
        loss_attn_B = self.criterion_attn(attn_B_up * feat_real_B, attn_B_up * feat_fake_A)
        
        loss_attn = (loss_attn_A + loss_attn_B) * 0.5
        
        # Total Generator Loss (Equation 5)
        loss_G = loss_GAN + self.lambda_cyc * loss_cyc + self.lambda_attn * loss_attn
        loss_G.backward()
        optimizer_G.step()
        
        # ------------------
        # Discriminator Step
        # ------------------
        optimizer_D.zero_grad()
        
        # D_A Loss
        pred_real_A = self.D_A(real_A)
        loss_D_real_A = self.criterion_GAN(pred_real_A, torch.ones_like(pred_real_A))
        pred_fake_A = self.D_A(fake_A.detach())
        loss_D_fake_A = self.criterion_GAN(pred_fake_A, torch.zeros_like(pred_fake_A))
        loss_D_A = (loss_D_real_A + loss_D_fake_A) * 0.5
        loss_D_A.backward()
        
        # D_B Loss
        pred_real_B = self.D_B(real_B)
        loss_D_real_B = self.criterion_GAN(pred_real_B, torch.ones_like(pred_real_B))
        pred_fake_B = self.D_B(fake_B.detach())
        loss_D_fake_B = self.criterion_GAN(pred_fake_B, torch.zeros_like(pred_fake_B))
        loss_D_B = (loss_D_real_B + loss_D_fake_B) * 0.5
        loss_D_B.backward()
        
        optimizer_D.step()
        
        return {
            'loss_G': loss_G.item(),
            'loss_D_A': loss_D_A.item(),
            'loss_D_B': loss_D_B.item(),
            'loss_cyc': loss_cyc.item(),
            'loss_attn': loss_attn.item()
        }

if __name__ == "__main__":
    print("Initializing Attention-CycleGAN...")
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    
    # Initialize model with paper hyperparameters (L_cyc=10, L_attn=5)
    model = AttentionCycleGAN(lambda_cyc=10.0, lambda_attn=5.0).to(device)
    
    # Adam Optimizers as specified in the paper (beta1=0.5, beta2=0.999)
    optimizer_G = optim.Adam(
        list(model.G_A2B.parameters()) + list(model.G_B2A.parameters()),
        lr=0.0002, betas=(0.5, 0.999)
    )
    optimizer_D = optim.Adam(
        list(model.D_A.parameters()) + list(model.D_B.parameters()),
        lr=0.0002, betas=(0.5, 0.999)
    )
    
    print("Generating dummy datasets...")
    # Dummy data mimicking a batch of 4 RGB images of size 256x256
    B, C, H, W = 4, 3, 256, 256
    # Domain A (Cultural Relics) and Domain B (Ink-wash)
    dummy_real_A = torch.randn(B, C, H, W).to(device) * 0.5 + 0.5 
    dummy_real_B = torch.randn(B, C, H, W).to(device) * 0.5 + 0.5
    
    print("Starting dummy training loop...")
    model.train()
    num_steps = 3
    for step in range(num_steps):
        # Rescale to [-1, 1] for Tanh output compatibility
        real_A = dummy_real_A * 2 - 1
        real_B = dummy_real_B * 2 - 1
        
        losses = model.train_step(real_A, real_B, optimizer_G, optimizer_D)
        
        print(f"Step [{step+1}/{num_steps}], "
              f"G Loss: {losses['loss_G']:.4f}, "
              f"D_A Loss: {losses['loss_D_A']:.4f}, "
              f"D_B Loss: {losses['loss_D_B']:.4f}, "
              f"Cyc Loss: {losses['loss_cyc']:.4f}, "
              f"Attn Loss: {losses['loss_attn']:.4f}")
        
    print("Execution complete. The model architecture and losses are correctly configured.")
